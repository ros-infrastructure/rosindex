root = sys.argv[1]

    rp = os.path.join(root, repo)
    if not os.path.isdir(rp): continue
    for d, dirs, files in os.walk(rp):
        dirs[:] = [x for x in dirs if not x.startswith('.')]
        ignored = any(f in files for f in ('COLCON_IGNORE','AMENT_IGNORE','CATKIN_IGNORE'))
        if 'package.xml' in files:
            p = os.path.join(d, 'package.xml')
            try:
                t = ET.parse(p).getroot()
                name = (t.findtext('name') or '').strip()
                lic = [{'text': (l.text or '').strip(), 'file': l.get('file')} for l in t.findall('license')]
                out.append({'repo': repo, 'path': os.path.relpath(p, root), 'name': name,
                            'format': t.get('format'), 'ignored': ignored, 'licenses': lic})
            except Exception as e:
                out.append({'repo': repo, 'path': os.path.relpath(p, root), 'error': str(e)})
            dirs[:] = []  # packages don't nest
json.dump(out, open(sys.argv[2], 'w'), indent=1)
print(len(out))
EOF
